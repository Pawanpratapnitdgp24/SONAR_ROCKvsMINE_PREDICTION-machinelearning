



SELECT 
IT.POLICY_NO,
IT.INCEPTION_DT,
PAL.SITE_NO,
PAL.LOCATION_ADDRESS_1_TX,
PAL.LOCATION_ADDRESS_2_TX,
PAL.LOCATION_ADDRESS_3_TX,
PAL.LOCATION_CITY_TX,
PAL.LOCATION_STATE_CD,
PAL.LOCATION_ZIP_CD,
TSC.RISK_STATE_COUNTRY_SK AS LOCATION_STATE_COUNTRY_SK ,
PAL.LOCATION_INSURED_BY_PCG_IN,
PAL.PROTECTION_CLASS_CD,
CASE 
WHEN IT.SOURCE_SYSTEM_NM='Instanda Admitted'THEN 20 
WHEN IT.SOURCE_SYSTEM_NM ='Instanda Non-Admitted' THEN 21 
ELSE 0
END AS SOURCE_SYSTEM_SK,
PAL.OCCUPANCY_TYPE_TX,
PAL.CONSTRUCTION_TYPE_TX,
PAL.BUILD_YR_NO,
PAL.SITE_NO,
PAL.MILES_FROM_SHORE_NO,
PAL.FEET_FROM_SHORE_NO,
PAL.SEA_SEC_SRCHRG_TX,
PAL.NUMBER_OF_FAMILIES_NO
from STAGE_MISQ.INSTANDA_PIM_ADDL_LOCATION PAL
inner join STAGE_MISQ.INSTANDA_TRANSACTION IT
on PAL.QUOTEREF=IT.QUOTEREF
AND IT.SaleStage IN ('SaleComplete', 'Cancelled', 'CancelFromRenewal',  'CancelScheduled', 'CancelReinstated')
--AND TRUNC(PAL.PCS_STG_LOAD_TS) = '10-FEB-25'
LEFT OUTER JOIN  MISQ.TRISK_STATE_COUNTRY TSC on
PAL.LOCATION_STATE_CD = TSC.RISK_STATE_CD 
LEFT OUTER JOIN MISQ.TPIM_LOCATION TL ON
PAL.POLICY_NO = TL.POLICY_NO ;
--PAL.INCEPTION_DT = TL.INCEPTION_DT ;
SELECT * FROM MISQ.TPIM_LOCATION;
SELECT DISTINCT * FROM MISQ.tpim_addl_location where trunc(trigger_last_update_ts) = '10-FEB-25';
SELECT * FROM STAGE_MISQ.INSTANDA_PIM_ADDL_LOCATION;
