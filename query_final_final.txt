SELECT 
    IQP.QUOTE_NO,
    IQP.INCEPTION_DT,
    TQ.QUOTE_SK,
    TQ.QUOTE_NO AS TQ_QUOTE_NO,
    TQ.POLICY_NO,
    TQ.UNDERWRITER,
    TQ.POLICY_TYPE,
    TQ.QUOTE_STATUS,
    TQ.BIND_DATE,
    TQD.COVERAGE_TYPE,
    TQD.PREMIUM_AMOUNT,
    TQD.PAYMENT_FREQUENCY,
    TQD.EFFECTIVE_DATE,
    TQD.EXPIRATION_DATE,
    CASE 
        WHEN IQP.SOURCE_SYSTEM_NM='Instanda Admitted' THEN 30 
        WHEN IQP.SOURCE_SYSTEM_NM='Instanda Non-Admitted' THEN 31 
        ELSE 0
    END AS SOURCE_SYSTEM_SK,
    IQP.POLICY_TYPE AS IQP_POLICY_TYPE,
    IQP.UNDERWRITER AS IQP_UNDERWRITER,
    IQP.QUOTE_STATUS AS IQP_QUOTE_STATUS,
    IQP.BIND_DATE AS IQP_BIND_DATE
FROM STAGE_MISQ.INSTANDA_QUOTE_POLICY_REFERENCE IQP
INNER JOIN MISQ.TQUOTE_POLICY_REFERENCE TQ
    ON IQP.QUOTE_NO = TQ.QUOTE_NO
    AND TQ.QUOTE_STATUS IN ('Bound', 'Issued', 'Declined', 'Cancelled')
LEFT OUTER JOIN MISQ.TQUOTE_POLICY_DETAILS TQD
    ON IQP.QUOTE_NO = TQD.QUOTE_NO;
