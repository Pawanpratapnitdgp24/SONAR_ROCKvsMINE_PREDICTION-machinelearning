SELECT
IPAL.BUILD_YR_NO ,
TRIM(IPAL.CONSTRUCTION_TYPE_TX) AS CONSTRUCTION_TYPE_TX,
TO_NUMBER(IPAL.FEET_FROM_SHORE_NO) AS FEET_FROM_SHORE_NO, 
CAST(TO_TIMESTAMP( IT.INCEPTION_DT , 'DD-MON-RR HH.MI.SS.FF6 AM') AS DATE) INCEPTION_DT,   
TRIM(IPAL.LOCATION_ADDRESS_1_TX ) AS LOCATION_ADDRESS_1_TX,
TRIM(IPAL.LOCATION_ADDRESS_2_TX) AS LOCATION_ADDRESS_2_TX ,
TRIM(IPAL.LOCATION_CITY_TX) AS LOCATION_CITY_TX,
TRIM(IPAL.LOCATION_STATE_CD) AS LOCATION_STATE_CD , 
TRIM(IPAL.LOCATION_STATE_COUNTRY) AS LOCATION_STATE_COUNTRY ,
TRIM(IPAL.LOCATION_ZIP_CD) AS LOCATION_ZIP_CD ,
TO_NUMBER(IPAL.MILES_FROM_SHORE_NO) AS MILES_FROM_SHORE_NO , 
TRIM(IPAL.NUMBER_OF_FAMILIES_NO) AS NUMBER_OF_FAMILIES_NO , 
TRIM(IPAL.OCCUPANCY_TYPE_TX) AS OCCUPANCY_TYPE_TX , 
IPAL.PIMADDLLOCATION_INDEX ,     
TRIM(IT.POLICY_NO) AS POLICY_NO ,
TRIM(IPAL.PROTECTION_CLASS_CD) AS PROTECTION_CLASS_CD , 
(IPAL.PIMADDLLOCATION_INDEX+1) AS SITE_NO,  
TRIM(IT.SOURCE_SYSTEM_NM) AS SOURCE_SYSTEM_NM,   
TRIM(IPAL.LOCATION_ADDRESS_3_TX) AS LOCATION_ADDRESS_3_TX ,
TRIM(IPAL.LOCATION_INSURED_BY_PCG_IN) AS LOCATION_INSURED_BY_PCG_IN,   
IPAL.SITE_NO AS ORIGINAL_SITE_NO,    
TRIM(IPAL.SEA_SEC_SRCHRG_TX) AS SEA_SEC_SRCHRG_TX,  
TRIM(IPAL.PRODUCT_TYPE) AS PRODUCT_TYPE,  
'1' as DUMMY_FIELD
 
FROM (SELECT * FROM #PS_PCG_EDW.$PCG_EDW_Stage_Schema#.INSTANDA_PIM_ADDL_LOCATION WHERE TRUNC(#ExtractionColumn#)  
BETWEEN TO_DATE(#StartDate# , 'YYYYMMDD') AND TO_DATE(#EndDate# , 'YYYYMMDD')) IPAL
INNER JOIN
#PS_PCG_EDW.$PCG_EDW_Stage_Schema#.INSTANDA_TRANSACTION IT
ON IPAL.QUOTEREF = IT.QUOTEREF AND IT.SaleStage IN ('SaleComplete', 'Cancelled', 'CancelFromRenewal',  'CancelScheduled', 'CancelReinstated');
