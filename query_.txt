WITH Staging AS (
    SELECT
        POLICY_NO AS Staging_POLICY_NO,
        INCEPTION_DT AS Staging_INCEPTION_DT,
        SITE_NO AS Staging_SITE_NO,
        LOCATION_ADDRESS_1_TX AS Staging_LOCATION_ADDRESS_1_TX,
        LOCATION_ADDRESS_2_TX AS Staging_LOCATION_ADDRESS_2_TX,
        LOCATION_ADDRESS_3_TX AS Staging_LOCATION_ADDRESS_3_TX,
        LOCATION_CITY_TX AS Staging_LOCATION_CITY_TX,
        LOCATION_STATE_CD AS Staging_LOCATION_STATE_CD,
        LOCATION_ZIP_CD AS Staging_LOCATION_ZIP_CD,
        LOCATION_INSURED_BY_PCG_IN AS Staging_LOCATION_INSURED_BY_PCG_IN,
        PROTECTION_CLASS_CD AS Staging_PROTECTION_CLASS_CD,
        SOURCE_SYSTEM_NM AS Staging_SOURCE_SYSTEM_NM,
        OCCUPANCY_TYPE_TX AS Staging_OCCUPANCY_TYPE_TX,
        CONSTRUCTION_TYPE_TX AS Staging_CONSTRUCTION_TYPE_TX,
        BUILD_YR_NO AS Staging_BUILD_YR_NO,
        ORIGINAL_SITE_NO AS Staging_ORIGINAL_SITE_NO,
        MILES_FROM_SHORE_NO AS Staging_MILES_FROM_SHORE_NO,
        FEET_FROM_SHORE_NO AS Staging_FEET_FROM_SHORE_NO,
        NUMBER_OF_FAMILIES_NO AS Staging_NUMBER_OF_FAMILIES_NO,
        CURRENT_TIMESTAMP() AS TRIGGER_LAST_UPDATE_TS
    FROM INSTANDA_PIM_ADDL_LOCATION
)

SELECT
    'TPIM_ADDL_LOCATION' AS DWH_Table_Name,
    Staging.Staging_POLICY_NO AS POLICY_NO,
    RIGHT(Staging.Staging_POLICY_NO, 10) AS PolicyNumber,
    CAST(TO_TIMESTAMP(Staging.Staging_INCEPTION_DT, 'DD-MON-RR HH.MI.SS.FF6 AM') AS DATE) AS InceptionDate_DATE,
    Staging.Staging_SITE_NO AS SITE_NO,
    TRIM(Staging.Staging_LOCATION_ADDRESS_1_TX) AS CO_SecondaryAddressLine1_TXT,
    TRIM(Staging.Staging_LOCATION_ADDRESS_2_TX) AS CO_SecondaryAddressLine2_TXT,
    TRIM(Staging.Staging_LOCATION_ADDRESS_3_TX) AS CO_SecondaryAddress3_TXT,
    TRIM(Staging.Staging_LOCATION_CITY_TX) AS CO_SecondaryAddressCity_TXT,
    TRIM(Staging.Staging_LOCATION_STATE_CD) AS CO_SecondaryAddressState_TXT,
    TRIM(Staging.Staging_LOCATION_ZIP_CD) AS CO_SecondaryAddressPostcode_TXT,
    NULLIF(Staging.Staging_LOCATION_INSURED_BY_PCG_IN, '') AS LOCATION_INSURED_BY_PCG_IN,
    TRIM(Staging.Staging_PROTECTION_CLASS_CD) AS CO_SProtectionClass_CHOICE,
    CASE 
        WHEN Staging.Staging_SOURCE_SYSTEM_NM = 'Instanda Admitted' THEN 20
        WHEN Staging.Staging_SOURCE_SYSTEM_NM = 'Instanda Non-Admitted' THEN 21
        ELSE 0
    END AS SOURCE_SYSTEM_NM,
    TRIM(Staging.Staging_OCCUPANCY_TYPE_TX) AS CO_SLocationType_CHOICE,
    TRIM(Staging.Staging_CONSTRUCTION_TYPE_TX) AS CO_SPrimary_ConstructionType_CHOICE,
    Staging.Staging_BUILD_YR_NO AS CO_SYearBuilt_NUM,
    Staging.Staging_ORIGINAL_SITE_NO AS ORIGINAL_SITE_NO,
    TO_NUMBER(Staging.Staging_MILES_FROM_SHORE_NO) AS MILES_FROM_SHORE_NO_SEC,
    TO_NUMBER(Staging.Staging_FEET_FROM_SHORE_NO) AS FEET_FROM_SHORE_NO_SEC,
    TRIM(Staging.Staging_NUMBER_OF_FAMILIES_NO) AS CO_SNumberofFamiliesinHome_CHOICE,
    Staging.TRIGGER_LAST_UPDATE_TS AS TRIGGER_LAST_UPDATE_TS
FROM Staging;
